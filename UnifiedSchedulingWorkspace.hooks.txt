    if (!notesStr) return {};
    const match = notesStr.match(/Form Data:\s*([\s\S]+)/);

    try {
      const raw = JSON.parse(match[1]);
      const clean: Record<string, any> = {};
      for (const [k, v] of Object.entries(raw)) {
        const cleanKey = k.replace(/[\r\n\t]+/g, ' ').replace(/\s+/g, ' ').replace(/\*/g, '').trim();
        clean[cleanKey] = v;
      }
      return clean;
    } catch {
      return {};
